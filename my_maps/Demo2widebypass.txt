# Le débit effectif du détour est divisé par 2 (500 -> 250), mais
# ça reste largement mieux que la route directe (capacité 1) --
# Yen doit s'activer malgré la pénalité RESTRICTED.

hub: C 2 -1 [zone=restricted max_drones=500]
hub: D 3 -1 [max_drones=500]
connection: start-A [max_link_capacity=500]
connection: A-B [max_link_capacity=1]
connection: B-goal [max_link_capacity=1]
