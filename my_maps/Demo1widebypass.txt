# Démo 1 : détour large, sans zone restreinte
# La route directe (start-A-B-goal) est étroite (capacité 1) et
# ne suffit jamais pour 500 drones. Le détour (A-C-D-E-goal) est
# large (capacité 500) mais plus long -- sans Yen, le DAG ne le
# verrait jamais puisque C est plus loin du but que A.
nb_drones: 500
start_hub: start 0 0 [max_drones=500]
hub: A 1 0 [max_drones=500]
hub: B 2 1 [max_drones=500]
end_hub: goal 3 0 [max_drones=500]
hub: C 2 -1 [max_drones=500]
hub: D 3 -1 [max_drones=500]
hub: E 4 -1 [max_drones=500]
connection: start-A [max_link_capacity=500]
connection: A-B [max_link_capacity=1]
connection: B-goal [max_link_capacity=1]
connection: A-C [max_link_capacity=500]
connection: C-D [max_link_capacity=500]
connection: D-E [max_link_capacity=500]
connection: E-goal [max_link_capacity=500]
